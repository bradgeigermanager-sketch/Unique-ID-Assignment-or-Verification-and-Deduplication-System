五、部署脚本
1️⃣ 部署脚本 (k8s/scripts/deploy.sh)
bash#!/bin/bash
set -e

# 颜色
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# 配置
NAMESPACE="${NAMESPACE:-org-governance}"
ENVIRONMENT="${ENVIRONMENT:-production}"
KUSTOMIZE_PATH="k8s/overlays/${ENVIRONMENT}"

echo -e "${GREEN}========================================${NC}"
echo -e "${GREEN}  组织治理系统部署脚本${NC}"
echo -e "${GREEN}========================================${NC}"

# 检查 kubectl
if ! command -v kubectl &> /dev/null; then
    echo -e "${RED}错误：kubectl 未安装${NC}"
    exit 1
fi

# 检查 kustomize
if ! command -v kustomize &> /dev/null; then
    echo -e "${RED}错误：kustomize 未安装${NC}"
    exit 1
fi

# 创建命名空间
echo -e "\n${YELLOW}1. 创建命名空间...${NC}"
kubectl create namespace ${NAMESPACE} --dry-run=client -o yaml | kubectl apply -f -

# 应用配置
echo -e "\n${YELLOW}2. 应用 Kustomize 配置...${NC}"
kustomize build ${KUSTOMIZE_PATH} | kubectl apply -f -

# 等待服务就绪
echo -e "\n${YELLOW}3. 等待服务就绪...${NC}"
sleep 10

# 检查后端
echo -e "\n${YELLOW}4. 检查后端服务...${NC}"
kubectl rollout status deployment/backend -n ${NAMESPACE} --timeout=300s

# 检查前端
echo -e "\n${YELLOW}5. 检查前端服务...${NC}"
kubectl rollout status deployment/frontend -n ${NAMESPACE} --timeout=300s

# 检查 Neo4j
echo -e "\n${YELLOW}6. 检查 Neo4j 服务...${NC}"
kubectl rollout status deployment/neo4j -n ${NAMESPACE} --timeout=300s

# 检查 PostgreSQL
echo -e "\n${YELLOW}7. 检查 PostgreSQL 服务...${NC}"
kubectl rollout status deployment/postgres -n ${NAMESPACE} --timeout=300s

# 显示状态
echo -e "\n${GREEN}========================================${NC}"
echo -e "${GREEN}  部署完成！${NC}"
echo -e "${GREEN}========================================${NC}"

echo -e "\n${YELLOW}服务状态:${NC}"
kubectl get pods -n ${NAMESPACE}

echo -e "\n${YELLOW}Ingress 信息:${NC}"
kubectl get ingress -n ${NAMESPACE}

echo -e "\n${YELLOW}访问地址:${NC}"
echo -e "  前端：https://org-governance.example.com"
echo -e "  API:  https://api.org-governance.example.com/api"
echo -e "  API 文档：https://api.org-governance.example.com/docs"
2️⃣ 回滚脚本 (k8s/scripts/rollback.sh)
bash#!/bin/bash
set -e

NAMESPACE="${NAMESPACE:-org-governance}"
REVISION="${REVISION:-}"

if [ -z "$REVISION" ]; then
    echo "用法：./rollback.sh <revision>"
    echo "示例：./rollback.sh 3"
    exit 1
fi

echo "回滚到版本 ${REVISION}..."

# 回滚后端
kubectl rollout undo deployment/backend -n ${NAMESPACE} --to-revision=${REVISION}

# 回滚前端
kubectl rollout undo deployment/frontend -n ${NAMESPACE} --to-revision=${REVISION}

# 等待回滚完成
kubectl rollout status deployment/backend -n ${NAMESPACE} --timeout=300s
kubectl rollout status deployment/frontend -n ${NAMESPACE} --timeout=300s

echo "回滚完成！"

2️⃣ 安装 Chart
bash# 安装到开发环境
helm install org-governance helm/org-governance \
  --namespace org-governance \
  --create-namespace \
  --set environment=development

# 安装到生产环境
helm install org-governance helm/org-governance \
  --namespace org-governance \
  --create-namespace \
  --set environment=production \
  --set backend.replicas=5 \
  --set frontend.replicas=3

# 升级
helm upgrade org-governance helm/org-governance \
  --namespace org-governance \
  --set backend.replicas=7

# 查看状态
helm status org-governance -n org-governance

# 卸载
helm uninstall org-governance -n org-governance

# 2. 配置 kubectl

# 3. 部署到开发环境

# 4. 部署到生产环境
export ENVIRONMENT=production
./k8s/scripts/deploy.sh

# 5. 验证部署

# 6. 查看日志
